File 1: src/components/portfoilo/Portfolio.a11y.test.tsx (unit tests with vitest-axe)

```typescript
import { describe, it, expect } from 'vitest'
import { render, screen } from '@testing-library/react'
import userEvent from '@testing-library/user-event'
import { axe } from 'vitest-axe'
import { MemoryRouter } from 'react-router-dom'
import Portfolio from './Portfolio'

// Helper: render with router context (needed for any link components)
const renderPortfolio = () => {
  return render(
    <MemoryRouter>
      <Portfolio />
    </MemoryRouter>
  )
}

describe('Portfolio - Accessibility (A11Y-11)', () => {
  it('has no accessibility violations on initial render', async () => {
    const { container } = renderPortfolio()
    const results = await axe(container)
    expect(results).toHaveNoViolations()
  })

  it('portfolio cards are keyboard accessible', () => {
    renderPortfolio()
    const cards = screen.getAllByRole('button')
    // Filter to portfolio cards (have aria-label with "project details")
    const portfolioCards = cards.filter(card =>
      card.getAttribute('aria-label')?.includes('project details')
    )
    expect(portfolioCards.length).toBeGreaterThan(0)
    portfolioCards.forEach(card => {
      expect(card).toHaveAttribute('tabIndex', '0')
    })
  })

  it('portfolio cards are wrapped in list structure', () => {
    renderPortfolio()
    const list = screen.getByRole('list')
    expect(list).toBeInTheDocument()
    const listItems = screen.getAllByRole('listitem')
    expect(listItems.length).toBeGreaterThan(0)
  })
})
```

NOTE on modal testing: The existing Portfolio.test.tsx already tests dialog open/close. The a11y test focuses on axe violations on initial render and card accessibility. Testing the open dialog with axe in jsdom is unreliable because jsdom doesn't implement HTMLDialogElement.showModal natively (it was mocked in Phase 2 tests). The E2E browser test (below) will test the open modal in a real browser.

```typescript
import Navbar from './Navbar'
```

File 3: src/test/a11y/e2e-accessibility.browser.test.tsx (E2E browser test with @axe-core/playwright)

This file uses the browser test config (vitest.browser.config.ts). It runs in a real Chromium browser.

```typescript
import { describe, it, expect } from 'vitest'
import { page } from '@vitest/browser-playwright'
import AxeBuilder from '@axe-core/playwright'

describe('E2E Accessibility (A11Y-10)', () => {
  it('homepage has no WCAG AA violations', async () => {
    // Navigate to the dev server
    await page.goto('http://localhost:5173/')

    // Wait for page to load
    await page.waitForSelector('#home')

    const results = await new AxeBuilder({ page: page.context().pages()[0] })
      .withTags(['wcag2a', 'wcag2aa', 'wcag21aa'])
      .analyze()

    // Log any violations for debugging
    if (results.violations.length > 0) {
      console.error('Accessibility violations:', JSON.stringify(results.violations, null, 2))
    }

    expect(results.violations).toEqual([])
  })
})
```

IMPORTANT: The E2E test requires a dev server running. The test runner should start the dev server or the test instructions should note `pnpm dev &` must be running. Check if vitest.browser.config.ts has a server config -- if not, the test may need adjustment. The browser test file goes in src/test/a11y/ to keep it organized separately from unit tests.

In Vitest browser mode with @vitest/browser-playwright, `page` might need to be accessed differently. import { page } from '@vitest/browser-playwright'

```typescript
describe('E2E Accessibility (A11Y-10)', () => {

    await page.goto('http://localhost:5173/')
    await page.waitForSelector('#home')

    // Inject and run axe-core directly
    const violations = await page.evaluate(async () => {
      const axe = await import('axe-core')
      const results = await axe.default.run(document, {
        runOnly: { type: 'tag', values: ['wcag2a', 'wcag2aa', 'wcag21aa'] }
      })
      return results.violations
    })

    expect(violations).toEqual([])
  })
})
```
