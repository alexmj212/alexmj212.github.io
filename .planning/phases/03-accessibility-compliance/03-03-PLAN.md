Add comprehensive accessibility styles to src/styles/base.css. Append these sections at the end of the file:

```css
/* ===== ACCESSIBILITY STYLES ===== */

/* Skip Link - visible only on focus (WCAG 2.4.1 Bypass Blocks) */
.skip-link {
  position: absolute;
  top: -100%;
  left: 16px;
  background: #000;
  color: #fff;
  padding: 8px 16px;
  text-decoration: none;
  z-index: 9999;
  border-radius: 0 0 4px 4px;
  font-weight: 600;
  font-size: 14px;
}

.skip-link:focus {
  top: 0;
  outline: 2px solid var(--carolina-blue);
  outline-offset: 2px;
}

.dark .skip-link {
  background: #1f2937;
  color: #f3f4f6;
}

.dark .skip-link:focus {
  outline-color: var(--carolina-blue-dark);
}

/* Global Focus Indicators (WCAG 2.4.13 Focus Appearance) */
/* Use focus-visible to show focus ring only for keyboard navigation, not mouse clicks */
*:focus-visible {
  outline: 2px solid var(--carolina-blue);
  outline-offset: 2px;
}

.dark *:focus-visible {
  outline-color: var(--carolina-blue-dark);
}

/* Remove outline for mouse clicks -- focus-visible handles keyboard-only */
*:focus:not(:focus-visible) {
  outline: none;
}

/* Reduced Motion (WCAG 2.3.3 Animation from Interactions) */
@media (prefers-reduced-motion: reduce) {
  *,
  *::before,
  *::after {
    animation-duration: 0.01ms !important;
    animation-iteration-count: 1 !important;
    transition-duration: 0.01ms !important;
    scroll-behavior: auto !important;
  }
}
```
